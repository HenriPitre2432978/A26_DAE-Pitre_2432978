
CREATE TABLE Auteur (
    IdAuteur INT AUTO_INCREMENT PRIMARY KEY,
    Nom VARCHAR(150) NOT NULL
);

CREATE TABLE Genre (
    IdGenre INT AUTO_INCREMENT PRIMARY KEY,
    Nom VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE Membre (
    IdMembre INT AUTO_INCREMENT PRIMARY KEY,
    Nom VARCHAR(150) NOT NULL,
    Email VARCHAR(254) NOT NULL UNIQUE,
    Telephone VARCHAR(25)
);

CREATE TABLE Livre (
    IdLivre INT AUTO_INCREMENT PRIMARY KEY,
    Titre VARCHAR(250) NOT NULL,
    DatePublication DATE NOT NULL,
    ISBN VARCHAR(20) NOT NULL UNIQUE,
    NbExemplaires INT NOT NULL CHECK (NbExemplaires >= 0),

    IdAuteur INT NOT NULL,
    IdGenre INT NOT NULL,

    CONSTRAINT FK_Livre_Auteur
        FOREIGN KEY (IdAuteur)
        REFERENCES Auteur(IdAuteur)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT FK_Livre_Genre
        FOREIGN KEY (IdGenre)
        REFERENCES Genre(IdGenre)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);

CREATE TABLE Emprunt (
    IdEmprunt INT AUTO_INCREMENT PRIMARY KEY,
    IdLivre INT NOT NULL,
    IdMembre INT NOT NULL,
    DateDebut DATE NOT NULL,
    
    /*Date+30 jrs = IA*/
    DateEcheance DATE
        GENERATED ALWAYS AS (DATE_ADD(DateDebut, INTERVAL 30 DAY)) STORED,

    DateRetour DATE DEFAULT NULL,

    CONSTRAINT FK_Emprunt_Livre
        FOREIGN KEY (IdLivre)
        REFERENCES Livre(IdLivre)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT FK_Emprunt_Membre
        FOREIGN KEY (IdMembre)
        REFERENCES Membre(IdMembre)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT CK_DateRetour
        CHECK (
            DateRetour IS NULL
            OR DateRetour >= DateDebut
        )
);
